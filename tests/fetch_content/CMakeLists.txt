cmake_minimum_required(VERSION 3.30)
project(mcpp_fetch_content_consumer LANGUAGES CXX)
include(CTest)
include(FetchContent)
option(MCPP_TEST_BEAST "Request the optional Beast interface" OFF)

# A parent's BUILD_TESTING=ON must not enable dependency tests and tools.
if(MCPP_TEST_BEAST)
  set(MCPP_WITH_BEAST ON CACHE BOOL "" FORCE)
  set(MCPP_WITH_BEAST_TLS "${MCPP_TEST_TLS}" CACHE BOOL "" FORCE)
endif()
if(DEFINED MCPP_GIT_REPOSITORY)
  FetchContent_Declare(MCPpp GIT_REPOSITORY "${MCPP_GIT_REPOSITORY}" GIT_TAG "${MCPP_GIT_TAG}")
else()
  FetchContent_Declare(MCPpp SOURCE_DIR "${MCPP_SOURCE_DIR}")
endif()
FetchContent_MakeAvailable(MCPpp)
if(MCPP_BUILD_TESTS OR MCPP_BUILD_EXAMPLES OR MCPP_INSTALL)
  message(FATAL_ERROR "FetchContent unexpectedly enabled MCP++ development targets")
endif()
foreach(target calculator fulfillment reflection_smoke)
  if(TARGET ${target})
    message(FATAL_ERROR "FetchContent added unexpected executable ${target}")
  endif()
endforeach()
if(MCPP_TEST_BEAST)
  find_package(Boost 1.92.0 EXACT REQUIRED COMPONENTS json)
  add_executable(fetch_beast ../consumer/beast.cpp ../consumer/second.cpp)
  target_link_libraries(fetch_beast PRIVATE mcp::beast Boost::json)
  target_compile_definitions(fetch_beast PRIVATE MCPP_EXPECT_TLS=$<BOOL:${MCPP_TEST_TLS}>)
  add_test(NAME fetch_beast_multiple_translation_units COMMAND fetch_beast)
else()
  add_executable(fetch_core ../consumer/core.cpp)
  target_link_libraries(fetch_core PRIVATE mcp::mcpp)
  add_test(NAME fetch_core COMMAND fetch_core)
endif()
